window.addBulkStockImeis=()=>{
  const text=$("#si_bulk").value;
  const vals=text.split(/\r?\n|,|;/).map(x=>x.trim()).filter(Boolean);
  for(const v of vals){
    const m=state.modal,receipt=m.receipt,it=receipt.items.find(x=>x.productId===m.productId);
    if(m.imeis.length>=it.qty)break;
    if(!m.imeis.includes(v)&&!db.imeis.some(i=>i.value===v))m.imeis.push(v);
  }
  render();
}
