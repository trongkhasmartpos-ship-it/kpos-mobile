/* ================= SCANNER ================= */
async function openScanner(target,context=null,continuous=false){
  state.scanTarget=target;state.scanContext=context;state.scannerContinuous=continuous;state.scanFeed=[];
  state.scanReturn=state.modal && state.modal.type!=="scanner"?state.modal:null;
  state.modal={type:"scanner",stream:null};render();setTimeout(startCameraScanner,100);
}
window.openScanner=openScanner;
async function startCameraScanner(){
  const video=$("#scannerVideo"),msg=$("#scannerMsg");if(!video)return;
  if(!navigator.mediaDevices?.getUserMedia){msg.textContent="Trình duyệt không hỗ trợ camera. Hãy nhập mã thủ công.";return}
  try{
    const stream=await navigator.mediaDevices.getUserMedia({video:{facingMode:{ideal:"environment"}}});
    video.srcObject=stream;state.modal.stream=stream;await video.play();
    if("BarcodeDetector" in window){
      let detector;try{detector=new BarcodeDetector({formats:["ean_13","ean_8","code_128","qr_code","upc_a","upc_e"]})}catch(e){detector=new BarcodeDetector()}
      let last="",lastAt=0;
      const loop=async()=>{
        if(state.modal?.type!=="scanner")return;
        try{
          const codes=await detector.detect(video);
          if(codes.length){
            const raw=codes[0].rawValue,now=Date.now();
            if(raw!==last || now-lastAt>1500){
              last=raw;lastAt=now;
              if(state.scannerContinuous){handleContinuousScan(raw)}else{acceptScan(raw);return}
            }
          }
        }catch(e){}
        requestAnimationFrame(loop);
      };loop();
    }else msg.textContent="Camera đang mở. Trình duyệt chưa hỗ trợ tự đọc barcode; có thể nhập mã thủ công.";
  }catch(e){msg.textContent="Không mở được camera. Hãy cấp quyền camera hoặc nhập mã thủ công."}
}

function handleContinuousScan(v){
  const exists=state.scanFeed.some(x=>x.code===v);
  if(exists){
    state.scanFeed.unshift({code:v,status:"Trùng",time:new Date().toISOString()});
    state.scanFeed=state.scanFeed.slice(0,20);renderScannerFeed();return;
  }
  let ok=false,label="";
  if(state.scanTarget==="stockContinuous"){
    const back=state.scanReturn;
    const p=findProductByCode(v);
    if(p&&back?.type==="stockReceipt"){
      let it=back.items.find(x=>x.productId===p.id);
      if(it)it.qty++;else back.items.push({productId:p.id,qty:1,imeis:[]});
      ok=true;label=p.name;
    }
  }else if(state.scanTarget==="imeiContinuous"){
    const back=state.scanReturn;
    if(back?.type==="stockImeis"){
      const receipt=back.receipt,it=receipt.items.find(x=>x.productId===back.productId);
      if(!db.imeis.some(i=>i.value===v)&&!back.imeis.includes(v)&&back.imeis.length<it.qty){
        back.imeis.push(v);ok=true;label="IMEI";
      }
    }
  }
  state.scanFeed.unshift({code:v,status:ok?"Hoàn thành":"Không nhận",label,time:new Date().toISOString()});
  state.scanFeed=state.scanFeed.slice(0,20);renderScannerFeed();
}
function renderScannerFeed(){
  const el=$("#scanFeed");if(!el)return;
  el.innerHTML=state.scanFeed.map(x=>`<div class="scan-row"><div><div class="scan-code">${esc(x.code)}</div><div class="small muted">${esc(x.label||"")}</div></div><div class="${x.status==="Hoàn thành"?"scan-ok":x.status==="Trùng"?"scan-dup":"danger-text"}">${x.status==="Hoàn thành"?"✓ Đã nhận":x.status}</div></div>`).join("");
}
window.finishContinuousScan=()=>{stopScanner();const back=state.scanReturn;state.scanReturn=null;state.scannerContinuous=false;state.modal=back||null;render()}

function stopScanner(){try{state.modal?.stream?.getTracks().forEach(t=>t.stop())}catch(e){}}
window.closeModal=()=>{stopScanner();state.modal=null;state.scanReturn=null;state.scanContext=null;render()}
window.acceptScan=(value=null)=>{
  const v=(value||$("#scannerManual")?.value||"").trim();if(!v)return;
  const target=state.scanTarget,back=state.scanReturn,ctx=state.scanContext;
  stopScanner();state.scanReturn=null;state.scanContext=null;

  if(target==="pos"){state.modal=null;const p=findProductByCode(v);if(p){addCart(p.id);return}render();toast("Không tìm thấy sản phẩm");return}
  if(target==="productSearch"){state.modal=null;state.screen="products";render();setTimeout(()=>{$("#productSearch").value=v;filterProducts()},0);return}
  if(target==="productForm"&&back?.type==="productForm"){back.p.barcode=v;state.modal=back;render();return}
  if(target==="imei"){state.modal=null;state.screen="imei";render();setTimeout(()=>{$("#imeiSearch").value=v;filterImeis()},0);return}
  if(target==="warrantySearch"){
    const im=db.imeis.find(i=>i.value===v);state.modal=null;
    if(im){openWarrantyForm(v);return}
    state.screen="warranty";render();setTimeout(()=>{$("#wSearch").value=v;filterWarranties()},0);return;
  }
  if(target==="warrantyForm"&&back?.type==="warrantyForm"){hydrateWarrantyDraft(back.draft,v);state.modal=back;render();return}
  if(target==="stock"){
    state.modal=back?.type==="stockReceipt"?back:{type:"stockReceipt",items:[],supplier:"",note:""};
    const p=findProductByCode(v);
    if(p){let it=state.modal.items.find(x=>x.productId===p.id);if(it)it.qty++;else state.modal.items.push({productId:p.id,qty:1,imeis:[]});render()}
    else{render();toast("Không tìm thấy sản phẩm")}return;
  }
  if(target==="stockImei"&&back?.type==="stockImeis"){state.modal=back;addStockImeiValue(v);return}
  state.modal=back||null;render();
}
