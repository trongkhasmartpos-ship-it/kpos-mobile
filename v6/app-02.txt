function crmDueList(mode="today"){
  const now=new Date();
  const start=new Date(now.getFullYear(),now.getMonth(),now.getDate());
  const end=new Date(start);end.setDate(end.getDate()+1);
  return db.customers.filter(c=>{
    if(!c.nextCareAt)return false;
    const d=new Date(c.nextCareAt);
    if(mode==="today")return d>=start&&d<end;
    if(mode==="overdue")return d<start&&c.status!=="Đã mua hàng"&&c.status!=="Không mua";
    if(mode==="upcoming")return d>=end;
    return true;
  }).sort((a,b)=>new Date(a.nextCareAt)-new Date(b.nextCareAt));
}
