function normalizeDB(raw){
  const out = {...clone(seed), ...raw};
  out.version = 2;
  ["products","customers","imeis","warranties","invoices","stockReceipts","payments","inventoryMovements","cart","activities","printTemplates"].forEach(k=>{
    if(!Array.isArray(out[k])) out[k]=[];
  });
  out.products = out.products.map(p=>({
    keywords:"",images:[],minStock:0,warranty:0,imeiEnabled:false,stock:0,cost:0,price:0,...p
  }));
  out.customers = out.customers.map(c=>({
    phone2:"",email:"",address:"",source:"",status:"Khách mới",total:0,debt:0,note:"",need:"",budget:"",nextCareAt:"",lastCareAt:"",careChannel:"Gọi điện",careHistory:[],...c
  }));
  out.imeis = out.imeis.map(i=>({
    stockReceiptId:null,customerId:null,invoiceId:null,receivedDate:null,soldDate:null,warrantyStart:null,warrantyEnd:null,...i
  }));
  out.printTemplates = (out.printTemplates||[]).length ? out.printTemplates : defaultPrintTemplates();
  out.cart = [];
  return out;
}

function loadDB(){
  try{
    const v2 = localStorage.getItem(DBKEY);
    if(v2) return normalizeDB(JSON.parse(v2));
    const old = localStorage.getItem(OLDKEY);
    if(old){
      const migrated = normalizeDB(JSON.parse(old));
      localStorage.setItem(DBKEY, JSON.stringify(migrated));
      return migrated;
    }
  }catch(e){}
  return clone(seed);
}

function money(n){return (Number(n)||0).toLocaleString("vi-VN")+"đ"}
function fmtDate(d=new Date()){
  if(!d) return "—";
  return new Intl.DateTimeFormat("vi-VN",{day:"2-digit",month:"2-digit",year:"numeric"}).format(new Date(d));
}
function fmtDateTime(d){
  if(!d) return "—";
  return new Intl.DateTimeFormat("vi-VN",{day:"2-digit",month:"2-digit",year:"numeric",hour:"2-digit",minute:"2-digit"}).format(new Date(d));
}
function addMonths(date,months){
  const d=new Date(date); d.setMonth(d.getMonth()+Number(months||0)); return d.toISOString();
}
function code(prefix, collection=[]){
  const d=new Date(), pad=n=>String(n).padStart(2,"0");
  const date=pad(d.getDate())+pad(d.getMonth()+1)+String(d.getFullYear()).slice(-2);
  const list=collection.filter(x=>String(x.code||"").startsWith(prefix+date));
  return `${prefix}${date}-${String(list.length+1).padStart(3,"0")}`;
}
